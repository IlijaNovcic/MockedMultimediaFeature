cmake_minimum_required(VERSION 3.20)
project(MockedMultimediaFeature CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(MockedMultimediaFeature 
src/main.cpp
)

target_include_directories(MockedMultimediaFeature PRIVATE src)
target_compile_options(MockedMultimediaFeature PRIVATE -Wall -Wextra -Wreorder)
